// --- FIREBASE CONFIG ---
const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : {};
const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';


// --- MAIN APP COMPONENT ---
export default function App() {
  const [db, setDb] = useState(null);
  const [auth, setAuth] = useState(null);
  const [userId, setUserId] = useState(null);
  const [isAuthReady, setIsAuthReady] = useState(false);

  const [level, setLevel] = useState(1);
  const [archetype, setArchetype] = useState('Entrepreneur');
  
  const [progress, setProgress] = useState({ blueprint: {}, kpis: {} });
  const [aiSuggestion, setAiSuggestion] = useState('');
  const [isLoadingAi, setIsLoadingAi] = useState(false);


  // Debounce saving to Firestore
  const debounce = (func, delay) => {
    let timeout;
    return (...args) => {
      clearTimeout(timeout);
      timeout = setTimeout(() => func(...args), delay);
    };
  };

  const saveProgressToFirestore = useCallback(
    async (newProgress) => {
        if (!db || !userId) return;
        try {
            const progressRef = doc(db, 'artifacts', appId, 'users', userId);
            await setDoc(progressRef, { polymath_progress: newProgress }, { merge: true });
        } catch (error) {
            console.error("Error saving progress to Firestore:", error);
        }
    }, [db, userId]
  );
  
  const debouncedSave = useCallback(debounce(saveProgressToFirestore, 1000), [saveProgressToFirestore]);


  // Initialize Firebase
  useEffect(() => {
    try {
      if (firebaseConfig.apiKey) {
        const app = initializeApp(firebaseConfig);
        const firestore = getFirestore(app);
        const fireAuth = getAuth(app);
        setDb(firestore);
        setAuth(fireAuth);
        setLogLevel('debug');
      }
    } catch (e) {
      console.error("Error initializing Firebase:", e);
    }
  }, []);

  // Handle Authentication
  useEffect(() => {
    if (!auth) return;
    const unsubscribe = onAuthStateChanged(auth, async (user) => {
      if (user) {
        setUserId(user.uid);
      } else {
        try {
          const token = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
          if(token){
            await signInWithCustomToken(auth, token);
          } else {
            await signInAnonymously(auth);
          }
        } catch (error) {
          console.error("Error signing in:", error);
        }
      }
      setIsAuthReady(true);
    });
    return () => unsubscribe();
  }, [auth]);

  // Fetch/Listen for Progress Data from Firestore
  useEffect(() => {
    if (!isAuthReady || !db || !userId) return;

    const progressRef = doc(db, 'artifacts', appId, 'users', userId);
    const unsubscribe = onSnapshot(progressRef, (docSnap) => {
      if (docSnap.exists() && docSnap.data().polymath_progress) {
        const existingProgress = docSnap.data().polymath_progress;
        setProgress({
            blueprint: existingProgress.blueprint || {},
            kpis: existingProgress.kpis || {}
        });
      } else {
        console.log("No progress document found. Starting fresh.");
        setProgress({ blueprint: {}, kpis: {} });
      }
    }, (error) => {
      console.error("Error listening to progress updates:", error);
    });

    return () => unsubscribe();
  }, [isAuthReady, db, userId]);

  const handleLevelChange = (newLevel) => {
    setLevel(newLevel);
    setArchetype('Entrepreneur');
    setAiSuggestion(''); // Clear AI suggestion on level change
  };

  const handleArchetypeChange = (newArchetype) => {
    setArchetype(newArchetype);
    setAiSuggestion(''); // Clear AI suggestion on archetype change
  }

  const handleBlueprintToggle = (blueprintItem) => {
    const newProgress = { ...progress };
    newProgress.blueprint[blueprintItem] = !newProgress.blueprint[blueprintItem];
    setProgress(newProgress);
    debouncedSave(newProgress);
  };

  const handleKpiChange = (kpiName, value) => {
    const newProgress = { ...progress };
    newProgress.kpis[kpiName] = value;
    setProgress(newProgress);
    debouncedSave(newProgress);
  };
  
  const getGeminiSuggestion = async () => {
    setIsLoadingAi(true);
    setAiSuggestion('');

    const currentData = canvasData[level].archetypes[archetype];
    
    const completedTasks = Object.keys(progress.blueprint || {}).filter(key => progress.blueprint[key] && currentData.blueprint.includes(key));
    const incompleteTasks = currentData.blueprint.filter(item => !completedTasks.includes(item));
    const kpiStatus = Object.entries(progress.kpis || {})
        .filter(([key]) => currentData.kpis.includes(key))
        .map(([key, value]) => `- ${key}: ${value || 'Not tracked yet'}`)
        .join('\n');

    const prompt = `
      As an expert coach, I need your advice. I am following a personal growth framework.
      
      My current status is:
      - Level: ${canvasData[level].level} (${canvasData[level].subtitle})
      - Archetype: ${archetype}
      - My Objective: ${currentData.objective}
      - My Mindset: ${currentData.mindset}

      My Progress:
      - Completed Blueprint Tasks:
      ${completedTasks.length > 0 ? completedTasks.map(task => `  - ${task}`).join('\n') : '  None yet.'}
      
      - Incomplete Blueprint Tasks:
      ${incompleteTasks.map(task => `  - ${task}`).join('\n')}

      - Current KPIs:
      ${kpiStatus || '  None yet.'}

      Based on this, please provide clear, actionable, and encouraging advice. Structure your response in Markdown with two sections:
      1.  **Analysis:** Briefly analyze my current situation.
      2.  **Next Steps:** Suggest 2-3 specific and actionable next steps I should take. Focus on the incomplete tasks and my KPIs.
    `;

    try {
        const chatHistory = [{ role: "user", parts: [{ text: prompt }] }];
        const payload = { contents: chatHistory };
        const apiKey = "";
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;
        
        const response = await fetch(apiUrl, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });

        if (!response.ok) {
            throw new Error(`API call failed with status: ${response.status}`);
        }
        
        const result = await response.json();
        
        if (result.candidates && result.candidates.length > 0 && result.candidates[0].content.parts.length > 0) {
            const text = result.candidates[0].content.parts[0].text;
            setAiSuggestion(text);
        } else {
            setAiSuggestion("Sorry, I couldn't generate a suggestion right now. Please try again.");
        }
    } catch (error) {
        console.error("Error calling Gemini API:", error);
        setAiSuggestion(`An error occurred: ${error.message}. Please check the console for details.`);
    } finally {
        setIsLoadingAi(false);
    }
  };


  const currentData = canvasData[level].archetypes[archetype];

  if (!isAuthReady) {
    return (
        <div className="flex items-center justify-center h-screen bg-gray-900 text-white">
            <div className="flex flex-col items-center">
                <BrainCircuit className="w-12 h-12 mb-4 animate-pulse text-indigo-400" />
                <p className="text-lg">Initializing Your Dashboard...</p>
            </div>
        </div>
    );
  }

  return (
    <div className="min-h-screen bg-gray-900 text-gray-200 font-sans p-4 sm:p-6 md:p-8">
      <div className="max-w-7xl mx-auto">
        <header className="text-center mb-8">
          <h1 className="text-4xl sm:text-5xl font-bold text-white tracking-tight">The Wealth Creation Canvas</h1>
          <p className="text-indigo-400 mt-2 text-lg">Your Interactive Blueprint for Growth</p>
        </header>

        {/* Level Selector */}
        <div className="flex justify-center mb-8 bg-gray-800/50 p-2 rounded-xl shadow-lg backdrop-blur-sm">
          {Object.keys(canvasData).map((lvl) => (
            <button
              key={lvl}
              onClick={() => handleLevelChange(lvl)}
              className={`px-3 py-2 sm:px-6 sm:py-3 text-sm sm:text-base font-semibold rounded-lg transition-all duration-300 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-offset-gray-900 focus:ring-indigo-500 mx-1 sm:mx-2
                ${level == lvl ? 'bg-indigo-600 text-white shadow-md' : 'text-gray-300 hover:bg-gray-700'}`}
            >
              Level {lvl}
            </button>
          ))}
        </div>

        <main className="bg-gray-800/50 rounded-2xl shadow-2xl p-4 sm:p-8 backdrop-blur-md border border-gray-700">
          <div className="text-center mb-6">
            <h2 className="text-3xl font-bold text-white">{canvasData[level].level}</h2>
            <p className="text-gray-400 mt-1">{canvasData[level].subtitle}</p>
          </div>

          {/* Archetype Tabs */}
          <div className="flex justify-center border-b border-gray-700 mb-6">
            {Object.keys(canvasData[level].archetypes).map((arch) => {
              const Icon = canvasData[level].archetypes[arch].icon;
              return (
              <button
                key={arch}
                onClick={() => handleArchetypeChange(arch)}
                className={`flex items-center px-4 py-3 font-medium border-b-2 transition-colors duration-300 focus:outline-none
                  ${archetype === arch ? 'border-indigo-500 text-indigo-400' : 'border-transparent text-gray-400 hover:text-white hover:border-gray-500'}`}
              >
                <Icon className="w-5 h-5 mr-2" />
                {arch}
              </button>
            )})}
          </div>

          <div className="space-y-8">
            {/* Main Content Grid */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
              <div className="space-y-6">
                <div className="bg-gray-800 p-6 rounded-xl border border-gray-700">
                  <h3 className="flex items-center text-xl font-semibold text-white mb-3"><Target className="w-6 h-6 mr-3 text-indigo-400" />Core Objective</h3>
                  <p className="text-gray-300">{currentData.objective}</p>
                </div>
                <div className="bg-gray-800 p-6 rounded-xl border border-gray-700">
                  <h3 className="flex items-center text-xl font-semibold text-white mb-3"><Brain className="w-6 h-6 mr-3 text-indigo-400" />Key Mindset Shift</h3>
                  <p className="text-gray-300 italic">"{currentData.mindset}"</p>
                </div>
                <div className="bg-gray-800 p-6 rounded-xl border border-gray-700">
                  <h3 className="flex items-center text-xl font-semibold text-white mb-4"><ListChecks className="w-6 h-6 mr-3 text-indigo-400" />Actionable Blueprint</h3>
                  <ul className="space-y-4">
                    {currentData.blueprint.map((item, index) => (
                      <li key={index} className="flex items-start">
                        <label className="flex items-center cursor-pointer">
                          <input type="checkbox" checked={!!progress.blueprint?.[item]} onChange={() => handleBlueprintToggle(item)} className="h-5 w-5 rounded bg-gray-700 border-gray-600 text-indigo-600 focus:ring-indigo-500 cursor-pointer" />
                          <span className={`ml-3 text-gray-300 ${progress.blueprint?.[item] ? 'line-through text-gray-500' : ''}`}>{item}</span>
                        </label>
                      </li>
                    ))}
                  </ul>
                </div>
              </div>
              <div className="bg-gray-800 p-6 rounded-xl border border-gray-700">
                <h3 className="flex items-center text-xl font-semibold text-white mb-4"><TrendingUp className="w-6 h-6 mr-3 text-indigo-400" />Key Performance Indicators</h3>
                <div className="space-y-4">
                  {currentData.kpis.map((kpi, index) => (
                    <div key={index}>
                      <label className="block text-sm font-medium text-gray-300 mb-1">{kpi}</label>
                      <input type="text" value={progress.kpis?.[kpi] || ''} onChange={(e) => handleKpiChange(kpi, e.target.value)} placeholder="Track your metric..." className="w-full bg-gray-700 border border-gray-600 rounded-md shadow-sm py-2 px-3 text-white focus:outline-none focus:ring-indigo-500 focus:border-indigo-500" />
                    </div>
                  ))}
                </div>
              </div>
            </div>

            {/* AI Suggestion Area */}
            <div className="bg-indigo-900/20 border border-indigo-800 rounded-xl p-6">
                 <div className="flex justify-between items-center mb-4">
                    <h3 className="text-xl font-semibold text-white flex items-center">
                        <Sparkles className="w-6 h-6 mr-3 text-indigo-400" />
                        AI-Powered Next Steps
                    </h3>
                    <button 
                        onClick={getGeminiSuggestion}
                        disabled={isLoadingAi}
                        className="bg-indigo-600 text-white font-bold py-2 px-4 rounded-lg hover:bg-indigo-700 transition duration-300 disabled:bg-indigo-800 disabled:cursor-not-allowed flex items-center"
                    >
                        {isLoadingAi ? (
                            <><BrainCircuit className="w-5 h-5 mr-2 animate-spin" /> Generating...</>
                        ) : (
                            <>✨ Get Suggestion</>
                        )}
                    </button>
                 </div>
                 {aiSuggestion && (
                    <div className="prose prose-invert max-w-none text-gray-300" dangerouslySetInnerHTML={{ __html: aiSuggestion.replace(/\n/g, '<br />') }} />
                 )}
            </div>
          </div>
        </main>
        <footer className="text-center mt-8 text-sm text-gray-500">
          <p>Your User ID: {userId}</p>
          <p>Built to help you achieve your goals. Keep pushing forward.</p>
        </footer>
      </div>
    </div>
  );
}
